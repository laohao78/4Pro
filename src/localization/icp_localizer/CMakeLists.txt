cmake_minimum_required(VERSION 3.0.2)
project(icp_localizer)
set(CMAKE_CXX_STANDARD 17)
find_package(catkin REQUIRED COMPONENTS roscpp pcl_ros pcl_conversions tf2_ros tf2_geometry_msgs sensor_msgs geometry_msgs nav_msgs)
find_package(PCL REQUIRED)
catkin_package(INCLUDE_DIRS include CATKIN_DEPENDS roscpp pcl_ros pcl_conversions tf2_ros tf2_geometry_msgs sensor_msgs geometry_msgs nav_msgs DEPENDS PCL)
include_directories(include ${catkin_INCLUDE_DIRS} ${PCL_INCLUDE_DIRS})
add_executable(icp_localizer_node src/icp_localizer_node.cpp src/icp_registration.cpp)
target_link_libraries(icp_localizer_node ${catkin_LIBRARIES} ${PCL_LIBRARIES})
install(TARGETS icp_localizer_node RUNTIME DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})
install(DIRECTORY launch config DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
